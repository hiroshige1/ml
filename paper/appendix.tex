\section{Derivations}
\label{app:deriv}

\paragraph{Hermite algebra.} For unit $u,w$ with $\langle u,w\rangle=m$ and $z_u=\langle u,x\rangle$, $z_w=\langle w,x\rangle$ jointly Gaussian with
correlation $m$, $\E[\sg_j(z_u)\sg_k(z_w)]=\delta_{jk}m^k$. For $\sg=\sg_2=(z^2-1)/\sqrt2$: $\sg_2^2=\sqrt6\,\sg_4+2\sqrt2\,\sg_2+1$, hence
$V(m)=\E[\sg_2(z_u)^2\sg_2(z_w)^2]=6m^4+8m^2+1$ and $g(m)=m^2$. For $\sg_1=z$: $g=m$, $V=1+2m^2$.

\paragraph{Population loss (Prop.~\ref{prop:loss}).} With $A=\frac1N\sum_iy_i\sg(\langle w,x_i\rangle)$ and $y_i=c\,\sg(\langle v,x_i\rangle)$,
$\E[A\mid c]=c\,g(m)$ and $\mathrm{Var}[A\mid c]=c^2(V(m)-g(m)^2)/N$. Since $x_q$ is independent of the context,
$\E[y_q\hat y_q]=\Gamma\,\E c^2\,g(m)^2$ and $\E[\hat y_q^2]=\Gamma^2g(1)\,\E c^2\,\big(g^2+(V-g^2)/N\big)$, giving~\eqref{eq:LA}.

\paragraph{Drift (Prop.~\ref{prop:repulsion}).} For $\sg_2$, $\Gamma=\gamma$, $s^2=1$:
$L_A=1-2\gamma m^4+\gamma^2\big[(1-\tfrac1N)m^4+(1+8m^2+6m^4)/N\big]$, so
$-\partial_mL_A=\big(8\gamma-4\gamma^2(1-\tfrac1N)-24\gamma^2/N\big)m^3-16\gamma^2m/N$, which is negative for $m<m^*$ with
$m^{*2}=16\gamma/\big(N(8-4\gamma(1-\tfrac1N)-24\gamma/N)\big)$. (An earlier draft of this derivation dropped the $24\gamma^2/N$ term and
had a factor 2 in the cubic coefficient; the closed-form loss, which the Monte-Carlo drift check verifies directly, was unaffected.)

\paragraph{General $\sg_k$.} The normalised second Hermite coefficient of $\sg_k^2$ is $\beta_2=\sqrt2\,k$ (checked by quadrature for
$k\le4$), so the finite-context repulsion is $-\Gamma^2\beta_2^2\,2m/N=-4k^2\Gamma^2m/N$ to leading order while the attraction is
$2\Gamma(2-\Gamma g(1))\,k\,m^{2k-1}$: the origin is drift-stable below $m^{*\,2(k-1)}\approx2k\Gamma/((2-\Gamma)N)$ for every $k\ge2$ (for $k=1$ the two
terms have the same power and the repulsion is a constant fraction). At $k=3$, $\Gamma=0.1$, $N=128$ the trap sets in at $d^*\approx17$
(against $\approx607$ at $k=2$), consistent with the negative Monte-Carlo drift observed for $k^*=3$ at $d=16$. The step size $\eta\propto d^{-2}$ is
the critical scaling only for $k^*=2$.

\paragraph{Tied $=$ pinned up to a time change.} The tied directional velocity equals the pinned velocity at $\gamma=\rho$ divided by $\rho$;
with $\dot\rho=-4\rho\,\partial_\rho L_A$ the pair $(m,\rho)$ above threshold obeys $m\propto t^{-2}$, $\rho\propto t^{-1}$ (LSODA to $t=10^6$), and
the moving threshold $\rho^*(m)\approx Nm^2/2\propto t^{-4}$ falls faster than $\rho$: the population flow traps permanently.

\paragraph{Escape-time scalings in Prop.~\ref{prop:shrink}.} (a)~With $\Gamma=\gamma$ fixed and $m_0>m^*$ the drift near the origin is
$\approx c_\gamma m^{2k^*-1}$ with $c_\gamma=\gamma(8-4\gamma)k^*\alpha_{k^*}^4/2$ for $k^*=2$, so $\tau_{\rm esc}\asymp m_0^{-(2k^*-2)}/c_\gamma$ up to the $N$-dependent
correction $-16\gamma^2m/N$, which saturates as $N\to\infty$. (b)~Along $\Gamma^*$, for $g^2\ll V/N$,
$L_A(m,\Gamma^*)\approx s^2[1-N g(m)^4/(g(1)V(m))]$ and $V(m)\to\beta_0^2=g(1)^2$ as $m\to0$, so the drift is $\approx s^2 N\,4k^*\alpha_{k^*}^8 m^{4k^*-1}/g(1)^3$:
$\tau_{\rm esc}\asymp m_0^{-(4k^*-2)}/N$, i.e.\ $\tau_{\rm esc}\propto d^{2k^*-1}/N$---the multiplicative $N$--$\tau$ trade-off. The condition for this
regime at the initial overlap is $m_0^{2k^*}\ll V/N$, i.e.\ $d^{k^*}\gg N$; for $d^{k^*}\ll N$ the signal-dominated form $1-g^2/g(1)$ applies and
$\tau_{\rm esc}\asymp m_0^{-(2k^*-2)}$ is $N$-independent. (c)~For the tied readout the attraction term $4gg'$ carries no $\rho$, so
$\tau_{\rm esc}\asymp m_0^{-(2k^*-2)}/(4k^*\alpha_{k^*}^4)$ independently of $N$ as long as $\rho\,V'/N\ll 4gg'$ along the trajectory, which holds
from $\rho_0\le\rho^*(m_0)/4$ (Theorem~\ref{thm:A}(iii)(c), with the $N$-dependence bounded by $C_k k^*\rho_0/(Nm_0^{2k^*-2})$), which for $k^*=2$ reads $\rho_0\ll Nm_0^2/2$; for $k^*=3$, $\rho_{\rm trap}\propto m_0^4$ and $\rho_0=0.01$ already violates it at $N=16$, $d\ge16$ (Exp.~9). At $d=512$, $k^*=2$ the three regimes give $\tau\propto\{\infty\text{ or }N\text{-saturating},\;N^{-1},\;N^0\}$, matching the
integrated flows ($\infty/745/379$; $8.9\cdot10^4/2.2\cdot10^4/5.3\cdot10^3$; $39/34/33$ for $N=32/128/512$).

\paragraph{Population ODEs (Prop.~\ref{prop:shrink}).} Spherical gradient flow for the overlap and plain flow for the readout:
free readout $\dot m=-(1-m^2)\partial_mL_A$, $\dot\Gamma=-r\,\partial_\Gamma L_A$ with $r=\eta_\Gamma/\eta$; tied readout
$\dot m=(1-m^2)\big[4gg'-\rho(2(1-\tfrac1N)gg'+V'/N)\big]$, $\dot\rho=-4\rho\,\partial_\rho L_A$. The chain rule through $\hat u=u/\|u\|$
gives the directional velocity a factor $1/\|u\|^2$ that cancels the $\rho=\|u\|^2$ carried by the attraction term but not the $\rho$ carried by the repulsion.
Escape times $T_{0.5}$ to $m=0.5$ from $m_0=c\,d^{-1/2}$ were integrated with an adaptive step ($1\%$ relative change per step) and
the exponent read off as $\kappa_{\rm eff}=2-\partial\log T/\partial\log m_0$ over $c\in\{0.5,0.7,1,1.4\}$, exactly as for the SGD data.

\paragraph{Kramers estimate.} $D_{\rm eff}=\eta\,\mathrm{sd}^2/2$ with the per-step standard deviation of the $m$-component of the batch gradient measured
by Monte Carlo ($0.33$--$0.37$ at $B=32$); the estimate ignores the spherical correction term, which is of the same order as the
quoted $d=32$ ratio, so only the $d=64$ value ($\approx20$) is meaningful.

\paragraph{Monte-Carlo drift check.} $d=16$, $N=128$, $\gamma=0.1$, $a=1$, $5\cdot10^4$ batches of $64$ prompts per point.
Ratios MC/formula at $m=0.2$ and $0.4$: Model B $k^*=1,2,3$: $0.998$--$1.00$; Model A $k^*=1$: $1.000$; $k^*=2$: $0.972$, $0.998$;
$k^*=3$: $0.88\pm0.76$ (drift $\approx0$), $1.02$. Model A $k^*=3$ at $\gamma=0.1$, $N=128$ has negative drift for $m\le0.2$.

SGD with fresh data each step, $B=32$ prompts per step unless stated, $\eta=\eta_0/d^2$. Projected samplers (exact in distribution)

$a=1$ (B). Slopes of $\log T_{0.5}$ vs $\log d$: B $k^*=1$: $1.98\pm0.06$ ($d\ge16$; the pre-registered $1.5$ was an error: for $\kappa<2$ the drift
